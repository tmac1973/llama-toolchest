It sits inside a line of text, so the template trims its own whitespace. */}} {{define "service_badge" -}} {{if eq .State "running"}}Running {{- else if eq .State "starting"}}Starting... {{- else if eq .State "failed"}}Failed {{.Error}} {{- else}}Stopped {{- end}} {{- with .Uptime}} ({{.}}){{end}} {{- end}} {{/* loaded_models lists the models the running router knows, with a filled dot for a loaded one. Data: []process.ModelStatus, not empty. */}} {{define "loaded_models" -}}
Models: {{- range .}} {{- if eq .Status.Value "loaded"}}
  ● {{.ID}} {{- else if eq .Status.Value "loading"}}
  ● {{.ID}} loading {{- else}}
  ○ {{.ID}} {{- end}} {{- end -}}
{{- end}}
